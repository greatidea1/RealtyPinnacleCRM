   186→export function PropertyDetailPage() {
   187→  const { user, selectedId, navigate, goBack, openPropertyForm, openDeleteDialog } = useAppStore();

   192→
   193→  useEffect(() => {
   194→    if (!selectedId) return;
   195→    Promise.all([
   196→      fetch(`/api/properties?id=${selectedId}&userId=${user?.id}&role=${user?.role}`).then(r => r.json()),
