   133→export function PropertiesListPage() {
   134→  const { user, navigate, openPropertyForm, openDeleteDialog } = useAppStore();
   135→
   136→  const [properties, setProperties] = useState<Property[]>([]);
   137→  const [loading, setLoading] = useState(true);
   138→  const [search, setSearch] = useState('');
   139→  const [statusFilter, setStatusFilter] = useState('');
   140→  const [viewMode, setViewMode] = useState<'table' | 'grid'>('table');
   141→  const [page, setPage] = useState(1);
   142→  const [total, setTotal] = useState(0);
   143→  const [counts, setCounts] = useState({ All: 0, Active: 0, Pending: 0, Sold: 0 });
   144→

   146→    if (!user) return;
   147→    setLoading(true);
   148→    try {
   149→      const params = new URLSearchParams({
   150→        userId: user.id,
   151→        role: user.role,
   152→        page: String(page),
   153→        limit: String(PAGE_SIZE),
   154→      });
   155→      if (statusFilter) params.set('status', statusFilter);
   156→      if (search.trim()) params.set('search', search.trim());
   157→
   158→      const res = await fetch(`/api/properties?${params}`);
   159→      const data = await res.json();
   160→      if (res.ok) {
   161→        setProperties(data.properties || []);
   162→        setTotal(data.total || 0);
   163→        setCounts(data.counts || { All: 0, Active: 0, Pending: 0, Sold: 0 });
   164→      }
   165→    } catch {
   166→      // silent fail
   167→    } finally {
   168→      setLoading(false);
   169→    }
   170→  }, [user, search, statusFilter, page]);
   171→
   172→  useEffect(() => {
   173→    fetchProperties();
   174→  }, [fetchProperties]);
   175→
   176→  const totalPages = Math.ceil(total / PAGE_SIZE);
   177→
   178→  return (
   179→    <div className="space-y-6">
   180→      {/* Header */}
   181→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   182→        <div>
   183→          <h1 className="text-2xl font-bold text-gray-900">Properties</h1>
   184→          <p className="text-sm text-gray-500 mt-0.5">
   185→            Manage your property listings
   186→          </p>
   187→        </div>
   188→        <Button
   189→          onClick={() => openPropertyForm()}
   190→          className="gap-2 gradient-primary text-white border-0 shadow-lg shadow-emerald-200 hover:shadow-emerald-300"
   191→        >
   192→          <Plus className="w-4 h-4" />
   193→          Add Property
   194→        </Button>
   195→      </div>
   196→
   197→      {/* Toolbar */}
   198→      <div className="glass-card rounded-xl p-4">
   199→        <div className="flex flex-col lg:flex-row lg:items-center gap-4">
   200→          {/* Search */}
   201→          <div className="relative flex-1 max-w-md">
   202→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />

   204→              placeholder="Search properties by title, locality..."
   205→              value={search}
   206→              onChange={(e) => {
   207→                setSearch(e.target.value);
   208→                setPage(1);
   209→              }}
   210→              className="pl-9"
   211→            />
   212→          </div>
   213→
   214→          {/* Filter Chips */}
   215→          <div className="flex items-center gap-2 flex-wrap">
   216→            {STATUS_FILTERS.map((f) => (

   218→                key={f.value}
   219→                onClick={() => {
   220→                  setStatusFilter(f.value);

   223→                className={cn(
   224→                  'px-3 py-1.5 rounded-full text-xs font-medium transition-all badge-glossy',
   225→                  statusFilter === f.value
   226→                    ? 'gradient-primary text-white shadow-md shadow-emerald-200'
   227→                    : 'bg-emerald-50 text-emerald-700 hover:bg-emerald-100 border border-emerald-200'
   228→                )}
   229→              >
   230→                {f.label}
   231→                <span className="ml-1.5 opacity-80">
   232→                  {(f.value ? counts[f.value as PropertyStatus] : counts.All) ?? 0}
   233→                </span>

   235→            ))}
   236→          </div>
   237→
   238→          {/* View Toggle */}
   239→          <div className="flex items-center gap-1 bg-emerald-50 rounded-lg p-1">
   240→            <button
   241→              onClick={() => setViewMode('table')}
   242→              className={cn(

   244→                viewMode === 'table'

   246→                  : 'text-gray-400 hover:text-emerald-600'

   249→              <List className="w-4 h-4" />

   252→              onClick={() => setViewMode('grid')}

   254→                'p-1.5 rounded-md transition-all',

   264→      </div>

   266→      {/* Content */}
   267→      <AnimatePresence mode="wait">
   268→        {loading ? (
   269→          <motion.div
   270→            key="loading"

   275→            {viewMode === 'table' ? (
   276→              <PropertiesTableSkeleton />

   278→              <PropertiesGridSkeleton />

   280→          </motion.div>
   281→        ) : properties.length === 0 ? (
   282→          <motion.div
   283→            key="empty"

   292→            <h3 className="text-lg font-semibold text-gray-700 mb-1">

   294→            </h3>
   295→            <p className="text-sm text-gray-400 max-w-sm">
   296→              {search || statusFilter
   297→                ? 'Try adjusting your search or filters.'

   299→            </p>
   300→            {!search && !statusFilter && (

   328→            key="grid"

   332→          >
   333→            <PropertiesGrid
   334→              properties={properties}
   335→              isAdmin={user?.role === 'ADMIN'}
   336→              onView={(p) => navigate('property-detail', p.id)}

   346→        <div className="flex items-center justify-between">
   347→          <p className="text-sm text-gray-500">
   348→            Showing {(page - 1) * PAGE_SIZE + 1}–
   349→            {Math.min(page * PAGE_SIZE, total)} of {total} properties

   351→          <div className="flex items-center gap-2">
   352→            <Button
   353→              variant="outline"

   358→              <ChevronLeft className="w-4 h-4" />
   359→            </Button>
   360→            {Array.from({ length: Math.min(5, totalPages) }, (_, i) => {

   362→              if (totalPages <= 5) {
   363→                pageNum = i + 1;
   364→              } else if (page <= 3) {
   365→                pageNum = i + 1;
   366→              } else if (page >= totalPages - 2) {
   367→                pageNum = totalPages - 4 + i;
   368→              } else {
   369→                pageNum = page - 2 + i;
   370→              }
   371→              return (
   372→                <Button
   373→                  key={pageNum}
   374→                  variant={page === pageNum ? 'default' : 'outline'}
   375→                  size="icon"
   376→                  onClick={() => setPage(pageNum)}
   377→                  className={cn(
   378→                    page === pageNum && 'gradient-primary border-0'
   379→                  )}
   380→                >
   381→                  {pageNum}
   382→                </Button>
   383→              );
   384→            })}
   385→            <Button
   386→              variant="outline"
   387→              size="icon"
   388→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   389→              disabled={page === totalPages}
   390→            >
   391→              <ChevronRight className="w-4 h-4" />
   392→            </Button>
   393→          </div>
   394→        </div>
   395→      )}
   396→    </div>
